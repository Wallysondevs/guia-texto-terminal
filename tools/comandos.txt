### csv-problema
echo '7,"Souza, Ana",28' | awk -F, '{print $2}'

### csv-fpat
echo '7,"Souza, Ana",28' | awk -v FPAT='([^,]*)|("[^"]*")' '{print $2; print NF, "campos"}'

### csv-mawk
echo '7,"Souza, Ana",28' | mawk -v FPAT='([^,]*)|("[^"]*")' '{print $2}'

### csv-tabela
awk -F, '{printf "%-11s %-9s %-8s %10s %6s\n", $1, $2, $3, $4, $5}' vendas.csv | head -5

### csv-erro
printf '7,Ana,28\r\n' | awk -F, '{print ($3 == "28") ? "igual" : "diferente"}'

### csv-m1
awk -F, 'BEGIN {OFS = ","; print "vendedor,total"} $3 == "mochila" {print $2, $4 * $5}' vendas.csv
